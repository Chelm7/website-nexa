async function kirimPesanan(e) { e.preventDefault(); const btn = document.getElementById('btnKirim'); const statusEl = document.getElementById('statusPesan'); btn.disabled = true; statusEl.className = "text-center text-sm font-bold mt-2 text-sky-600"; statusEl.innerText = "Mengirim pesanan..."; try { // Gunakan URL lengkap atau path eksplisit const res = await fetch('/api/pesanan', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ nama: document.getElementById('namaPemesan').value, menu: document.getElementById('pilihanMenu').value, catatan: document.getElementById('catatanPesanan').value, waktu: new Date().toLocaleString('id-ID') }) }); const textResponse = await res.text(); let hasil; try { hasil = JSON.parse(textResponse); } catch (err) { throw new Error("Server mengembalikan bukan format JSON: " + textResponse.substring(0, 50)); } if (hasil.success) { statusEl.className = "text-center text-sm font-bold mt-2 text-emerald-600"; statusEl.innerText = "Pesanan berhasil dikirim ke server!"; document.getElementById('formPesan').reset(); } else { throw new Error(hasil.error || "Gagal"); } } catch (err) { statusEl.className = "text-center text-sm font-bold mt-2 text-red-600"; statusEl.innerText = "Gagal: " + err.message; } finally { btn.disabled = false; } }
