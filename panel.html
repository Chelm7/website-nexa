async function muatPesanan() { const container = document.getElementById('daftarPesanan'); container.innerHTML = '

Memuat data dari server...

'; try { const response = await fetch('/api/pesanan'); const textResponse = await response.text(); let data; try { data = JSON.parse(textResponse); } catch (err) { throw new Error("Server mengembalikan HTML/Teks, bukan JSON. Cek apakah server Node.js aktif."); } if (data.success && Array.isArray(data.pesanan) && data.pesanan.length > 0) { let html = ''; data.pesanan.forEach(item => { html += `

👤 ${item.nama || 'Tanpa Nama'}

${item.waktu || '-'}

Menu: ${item.menu || '-'}

Catatan: ${item.catatan || 'Tidak ada'}

`; }); container.innerHTML = html; } else { container.innerHTML = '

Belum ada pesanan masuk.

'; } } catch (err) { container.innerHTML = `

Gagal memuat: ${err.message}

`; } }
